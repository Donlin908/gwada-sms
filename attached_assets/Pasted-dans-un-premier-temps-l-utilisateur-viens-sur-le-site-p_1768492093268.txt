3) Micro-flask d’exemple (gratuit)
Python
Copy
from flask import Flask, render_template_string
import requests, re, lxml.html

app = Flask(__name__)

@app.route("/<pays>")  # pays = france  ou  united-states
def nums(pays):
    url = f"https://www.smsonline.cloud/fr/country/{pays.capitalize()}"
    doc = lxml.html.fromstring(requests.get(url, timeout=10).text)
    nums = doc.xpath('//a[contains(@href,"/number/")]/text()')
    return render_template_string("""
    <h2>Numéros {{pays}} disponibles maintenant</h2>
    <ul>{% for n in nums %}<li>{{n}}</li>{% endfor %}</ul>
    <p> Rafraîchir dans 30 s pour voir les SMS arriver.</p>
    """, nums=nums, pays=pays)

if __name__ == "__main__":
    app.run(port=5000)
L’utilisateur clique sur un numéro, le copie, le colle dans le service tiers, revient sur la page ; le code de vérification apparaît sans qu’il ait à répondre.
